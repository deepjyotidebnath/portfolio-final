from langchain import Retriever, LLM
from guardrails import Judge

# hybrid retrieval + grounded answers
retriever = Retriever(
    dense="pinecone", sparse="bm25",
    top_k=8, rerank=True
)

def answer(query: str):
    docs = retriever.search(query)
    out = LLM.generate(query, ctx=docs)
    return Judge.verify(out, docs)
